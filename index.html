async function generujAlbertPDF() { try { const vzor = await localforage.getItem('vzor_pdf_v51'); if(!vzor) { ukazToast("Chyba: Nahraj PDF vzor!", "error"); return; } const { PDFDocument, StandardFonts } = PDFLib; const pdfDoc = await PDFDocument.load(vzor); // ZAJIŠTĚNÍ FONTU - toto je klíčové, aby to nepadalo const font = await pdfDoc.embedFont(StandardFonts.HelveticaBold); const pages = pdfDoc.getPages(); const page = pages[0]; const storeId = document.getElementById('global-prodejna').value; const mes = document.getElementById('sel-mesic').value; // Kreslení textu page.drawText(storeId, { x: 220, y: 670, size: 10, font: font }); page.drawText(`${mes}/2026`, { x: 480, y: 670, size: 10, font: font }); // Data z deníku let db = JSON.parse(localStorage.getItem(DB_KEY) || "[]"); let storeLogs = db.filter(z => z.prodejna.includes(storeId)); let y = 560; storeLogs.slice(0, 5).forEach((z) => { page.drawText(z.datum, { x: 50, y: y, size: 8, font: font }); page.drawText(z.popis.substring(0, 30), { x: 170, y: y, size: 8, font: font }); page.drawText((z.trvani / 60).toFixed(1), { x: 530, y: y, size: 8, font: font }); y -= 90; }); // Podpis const sB64 = await localforage.getItem('podpis_v51'); if(sB64) { const pngImage = await pdfDoc.embedPng(sB64); page.drawImage(pngImage, { x: 280, y: 630, width: 70, height: 25 }); } const pdfBytes = await pdfDoc.save(); const blob = new Blob([pdfBytes], { type: 'application/pdf' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = "Vykaz.pdf"; document.body.appendChild(a); a.click(); document.body.removeChild(a); ukazToast("PDF vygenerováno!"); } catch(e) { ukazToast("Chyba: " + e.message, "error"); console.error(e); } }
